import random

while True:
    choice = input("1 — посчитать (формат: число операция число), 2 — случайное число, 3 — выход: ")
    if choice == "3":
        print("До встречи!")
        break

    if choice == "1":
        parts = input("Введи: число операция число (например, 5 + 3): ").split()
        if len(parts) != 3:
            print("Формат: число операция число")
            continue
        a, op, b = parts
        try:
            a, b = float(a), float(b)
            if op == "+": res = a + b
            elif op == "-": res = a - b
            elif op == "*": res = a * b
            elif op == "/":
                res = a / b if b != 0 else "Ошибка: деление на ноль"
            else:
                print("Неизвестная операция")
                continue
            print("Результат:", res)
        except ValueError:
            print("Вводи числа корректно.")

    elif choice == "2":
        print("Случайное число:", random.randint(1, 100))
    else:
        print("Выбери 1, 2 или 3.")
